   110→/*  Placeholder con máquina de escribir (barra sin foco)               */
   111→/* ------------------------------------------------------------------ */
   112→function useTypewriter(active: boolean) {
   113→  const [text, setText] = useState("");
   114→  const stateRef = useRef({ phrase: 0, char: 0, deleting: false });
   115→
   116→  useEffect(() => {
   117→    if (!active) return;
   118→    const tick = () => {
   119→      const s = stateRef.current;
   120→      const phrase = TYPE_PHRASES[s.phrase % TYPE_PHRASES.length];
   121→      if (!s.deleting) {
   122→        s.char += 1;
   123→        if (s.char >= phrase.length) {
   124→          s.deleting = true;
   125→          return 1700;
   126→        }
   127→        return 48;
   128→      }
   129→      s.char -= 1;
   130→      if (s.char <= 0) {
   131→        s.deleting = false;
   132→        s.phrase += 1;
   133→        return 420;
   134→      }
   135→      return 26;
   136→    };
   137→
   138→    let timeout: ReturnType<typeof setTimeout>;
   139→    const loop = () => {
   140→      const delay = tick();
   141→      const s = stateRef.current;
   142→      setText(
   143→        TYPE_PHRASES[s.phrase % TYPE_PHRASES.length].slice(0, s.char)
   144→      );
   145→      timeout = setTimeout(loop, delay);
   146→    };
   147→    loop();
   148→    return () => clearTimeout(timeout);
   149→  }, [active]);
   150→
   151→  return text;
   152→}

   202→    const barW = bar.getBoundingClientRect().width;

   204→      g.respW !== el.offsetWidth || g.vw !== vw || g.barW !== barW

   214→
   215→      setPhase("thinking");
   216→      setDegraded(false);
   217→      setLastQuery(trimmed);
   218→      setFlight({ text: trimmed, stage: "in" });
   219→      try {
   220→        const res = await fetch("/api/homy/agent", {
   221→          method: "POST",
   222→          headers: { "Content-Type": "application/json" },
   223→          body: JSON.stringify({
   224→            message: trimmed,
   225→            mode: "auto",
   226→            lat: lat ?? null,
   227→            lng: lng ?? null,
   228→          }),
   229→        });

   231→        if (!res.ok || !data.ok) {
   232→          setReply({
   233→            ok: false,
   234→            message:
   235→              "El superagente tuvo un problema para razonar tu pedido. Probá de nuevo en unos segundos que sigo acá.",
   236→            suggestions: [],
   237→          });
   238→          setDegraded(true);
   239→        } else {
   240→          setReply(data);
   241→        }
   242→      } catch {
   243→        setReply({
   244→          ok: false,
   245→          message:
   246→            "No pude conectarme con el motor en este momento. Revisá tu conexión y volvé a intentar: sigo acá, cuidando tu hogar.",
   247→          suggestions: [],
   248→        });
   249→        setDegraded(true);
   250→      } finally {
   251→        setPhase("done");
   252→        setValue("");
   253→        inputRef.current?.blur();
   254→        setFlight((f) => (f ? { ...f, stage: "out" } : null));

   274→  };

   276→  const onSubmit = (e: FormEvent) => {

   281→
   282→  useEffect(

   284→      if (debounceRef.current) clearTimeout(debounceRef.current);

   287→    },
   288→    []
   289→  );
   290→
   291→  // prefers-reduced-motion: la demo ambiental no corre si el usuario la pide
   292→  useEffect(() => {
   293→    const mq = window.matchMedia("(prefers-reduced-motion: reduce)");

   295→    const onChange = (e: MediaQueryListEvent) => setReducedMotion(e.matches);
   296→    mq.addEventListener("change", onChange);
