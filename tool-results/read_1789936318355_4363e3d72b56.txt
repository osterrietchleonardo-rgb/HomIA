     1→'use client'
     2→// Materiales del cliente — marketplace de insumos sin contratar a nadie:
     3→// buscás lo que necesitás, ves TODAS las ofertas con precio, stock, reseñas y
     4→// distancia del proveedor, pedís el producto (abre el chat) y seguís tus
     5→// compras hasta pagar (Mercado Pago o efectivo) y reseñar al proveedor.
     6→import { useCallback, useEffect, useRef, useState } from 'react'
     7→import { navigate, useRoute } from '@/lib/router'
     8→import { useLocation } from '@/lib/store'
     9→import { Loading, EmptyState, UAvatar, UStars, VerifyBadge } from '@/components/app/ui-bits'
    10→import { formatARS } from '@/lib/format'
    11→import { formatDistance } from '@/lib/geo'
    12→import { toast } from 'sonner'
    13→import {
    14→  Search, Package, ShoppingBag, Store, MapPin, X, Star, Send, MessageCircle,
    15→  Banknote, CircleCheck, Hourglass, Undo2, Truck, Boxes,
    16→} from 'lucide-react'
    17→
    18→type Offer = {
    19→  stockId: string; elementId: string; price: number; quantity: number; brand: string | null
    20→  providerId: string; businessName: string; kind: string
    21→  providerUserId: string; providerAvatar: string | null; providerCity: string | null
    22→  providerRating: number; providerReviews: number; providerVerified: boolean
    23→  planPro: boolean; distanceKm?: number
    24→}
    25→type Result = {
    26→  elementId: string; name: string; description: string; aliases: string[]; unit: string
    27→  categorySlug: string; categoryName: string
    28→  offersCount: number; minPrice: number | null; maxPrice: number | null; hasPro: boolean
    29→  offers: Offer[]
    30→}
    31→type Purchase = {
    32→  id: string; elementName: string; quantity: number; unit: string; total: number
    33→  status: string; note?: string | null; chargeId?: string | null; createdAt: string
    34→  provider?: { id: string; businessName: string; userId: string }
    35→  client?: { id: string; displayName: string }
    36→}
    37→
    38→const CATEGORY_TABS = [
    39→  { slug: '', name: 'Todo' }, { slug: 'plomeria', name: 'Plomería' },
    40→  { slug: 'gasistas', name: 'Gas' }, { slug: 'electricistas', name: 'Electricidad' },
    41→  { slug: 'albanileria', name: 'Albañilería' }, { slug: 'durlock', name: 'Durlock' },
    42→  { slug: 'pintura', name: 'Pintura' }, { slug: 'herreria', name: 'Ferretería' },
    43→  { slug: 'herramientas', name: 'Herramientas' }, { slug: 'maderera', name: 'Maderera' },
    44→  { slug: 'carpinteria', name: 'Carpintería' }, { slug: 'techos', name: 'Techos' },
    45→  { slug: 'cerramientos', name: 'Aberturas' }, { slug: 'pisos', name: 'Pisos' },
    46→  { slug: 'aislacion', name: 'Aislación' }, { slug: 'iluminacion', name: 'Iluminación' },
    47→  { slug: 'climatizacion', name: 'Clima' }, { slug: 'jardineria', name: 'Jardín' },
    48→  { slug: 'limpieza', name: 'Limpieza' }, { slug: 'muebles', name: 'Muebles' },
    49→  { slug: 'seguridad', name: 'Seguridad' },
    50→]
    51→
    52→const STATUS_META: Record<string, { label: string; icon: typeof Hourglass; tone: string }> = {
    53→  solicitado: { label: 'Pedido enviado', icon: Hourglass, tone: 'text-[#B98A00] bg-[#FFC700]/12 ring-[#FFC700]/35' },
    54→  aceptado: { label: 'Aceptado por el proveedor', icon: CircleCheck, tone: 'text-[#1D63B8] bg-[#1D63B8]/10 ring-[#1D63B8]/30' },
    55→  entregado: { label: 'Listo para pagar', icon: Truck, tone: 'text-[#FF5A1F] bg-[#FF5A1F]/10 ring-[#FF5A1F]/30' },
    56→  pagado: { label: 'Pagado', icon: CircleCheck, tone: 'text-[#0e9f6e] bg-[#0e9f6e]/10 ring-[#0e9f6e]/30' },
    57→  cancelado: { label: 'Cancelado', icon: Undo2, tone: 'text-slate-500 bg-slate-500/10 ring-slate-400/30' },
    58→}
    59→
    60→export default function ClientMaterials() {
    61→  const route = useRoute()
    62→  const location = useLocation()
    63→  const [tab, setTab] = useState<'buscar' | 'compras'>(route.query.tab === 'compras' ? 'compras' : 'buscar')
    64→  const [q, setQ] = useState(route.query.q || '')
    65→  const [cat, setCat] = useState('')
    66→  const [results, setResults] = useState<Result[] | null>(null)
    67→  const [loading, setLoading] = useState(true)
    68→  const [purchases, setPurchases] = useState<Purchase[] | null>(null)
    69→  const [searched, setSearched] = useState(false)
    70→  const debounce = useRef<ReturnType<typeof setTimeout> | null>(null)
    71→
    72→  // diálogo de pedido
    73→  const [dlg, setDlg] = useState<{ offer: Offer; elementName: string; unit: string } | null>(null)
    74→  const [dlgQty, setDlgQty] = useState('1')
    75→  const [dlgNote, setDlgNote] = useState('')
    76→  const [busy, setBusy] = useState(false)
    77→  // diálogo de reseña
    78→  const [rv, setRv] = useState<Purchase | null>(null)
    79→  const [rvRating, setRvRating] = useState(0)
    80→  const [rvComment, setRvComment] = useState('')
    81→
    82→  const loadPurchases = useCallback(async () => {
    83→    try {
    84→      const res = await fetch('/api/purchases')
    85→      if (res.ok) setPurchases((await res.json()).purchases)
    86→    } catch { /* silencioso */ }
    87→  }, [])
    88→
    89→  const search = useCallback(async (query: string, category: string) => {
    90→    setLoading(true)
    91→    try {
    92→      const params = new URLSearchParams({ q: query, cat: category })
    93→      if (location.lat && location.lng) {
    94→        params.set('lat', String(location.lat))
    95→        params.set('lng', String(location.lng))
    96→        params.set('radius', String(location.radiusKm))
    97→      }
    98→      const res = await fetch(`/api/marketplace?${params}`)
    99→      if (res.ok) {
   100→        setResults((await res.json()).results)
   101→        setSearched(true)
   102→      }
   103→    } finally {
   104→      setLoading(false)
   105→    }
   106→  }, [location.lat, location.lng, location.radiusKm])
   107→
   108→  useEffect(() => {
   109→    if (tab !== 'buscar') return
   110→    if (debounce.current) clearTimeout(debounce.current)
   111→    debounce.current = setTimeout(() => void search(q, cat), 350)
   112→    return () => { if (debounce.current) clearTimeout(debounce.current) }
   113→  }, [q, cat, tab, search])
   114→
   115→  useEffect(() => {
   116→    if (tab === 'compras') loadPurchases()
   117→  }, [tab, loadPurchases])
   118→
   119→  async function pedir(e: React.FormEvent) {
   120→    e.preventDefault()
   121→    if (!dlg) return
   122→    const qty = parseFloat(dlgQty)
   123→    if (!qty || qty <= 0) { toast.error('Decinos cuánto necesitás'); return }
   124→    setBusy(true)
   125→    try {
   126→      const res = await fetch('/api/purchases', {
   127→        method: 'POST',
   128→        headers: { 'Content-Type': 'application/json' },
   129→        body: JSON.stringify({ stockId: dlg.offer.stockId, quantity: qty, note: dlgNote || undefined }),
   130→      })
   131→      const data = await res.json()
   132→      if (!res.ok) { toast.error(data.error || 'No se pudo enviar el pedido'); return }
   133→      toast.success('Pedido enviado', { description: `${dlg.elementName} × ${qty} ${dlg.unit} — te respondió el chat directo con ${dlg.offer.businessName}.` })
   134→      setDlg(null); setDlgQty('1'); setDlgNote('')
   135→      void loadPurchases()
   136→    } finally {
   137→      setBusy(false)
   138→    }
   139→  }
   140→
   141→  async function cancelar(p: Purchase) {
   142→    if (!confirm('¿Cancelar este pedido?')) return
   143→    const res = await fetch(`/api/purchases/${p.id}`, {
   144→      method: 'PATCH',
   145→      headers: { 'Content-Type': 'application/json' },
   146→      body: JSON.stringify({ action: 'cancelar' }),
   147→    })
   148→    const data = await res.json()
   149→    if (!res.ok) { toast.error(data.error); return }
   150→    toast.info('Pedido cancelado')
   151→    void loadPurchases()
   152→  }
   153→
   154→  async function pagarCharge(p: Purchase, method: 'mercadopago' | 'efectivo') {
   155→    if (!p.chargeId) return
   156→    setBusy(true)
   157→    try {
   158→      const res = await fetch(`/api/charges/${p.chargeId}`, {
   159→        method: 'POST',
   160→        headers: { 'Content-Type': 'application/json' },
   161→        body: JSON.stringify({ method }),
   162→      })
   163→      const data = await res.json()
   164→      if (!res.ok) {
   165→        if (data.needsConfig) toast.error('Mercado Pago no configurado en el servidor')
   166→        else toast.error(data.error)
   167→        return
   168→      }
   169→      if (method === 'mercadopago' && data.initPoint) {
   170→        window.open(data.initPoint, '_blank')
   171→        toast.info('Te abrimos Mercado Pago para completar el pago')
   172→      } else {
   173→        toast.success('Efectivo acordado', { description: 'El proveedor confirma cuando recibe el pago.' })
   174→      }
   175→      void loadPurchases()
   176→    } finally {
   177→      setBusy(false)
   178→    }
   179→  }
   180→
   181→  async function dejarReseña(e: React.FormEvent) {
   182→    e.preventDefault()
   183→    if (!rv || !rvRating) { toast.error('Elegí un puntaje de 1 a 5 estrellas'); return }
   184→    if (rvComment.trim().length < 10) { toast.error('Contanos un poco más (mínimo 10 caracteres)'); return }
   185→    setBusy(true)
   186→    try {
   187→      const res = await fetch('/api/reviews', {
   188→        method: 'POST',
   189→        headers: { 'Content-Type': 'application/json' },
   190→        body: JSON.stringify({
   191→          targetUserId: rv.provider?.userId,
   192→          rating: rvRating,
   193→          comment: rvComment.trim(),
   194→          context: 'compra',
   195→          purchaseId: rv.id,
   196→        }),
   197→      })
   198→      const data = await res.json()
   199→      if (!res.ok) { toast.error(data.error); return }
   200→      toast.success('¡Gracias! Tu reseña ayuda a toda la comunidad')
   201→      setRv(null); setRvRating(0); setRvComment('')
   202→      void loadPurchases()
   203→    } finally {
   204→      setBusy(false)
   205→    }
   206→  }
   207→
   208→  const hasReseña = async (p: Purchase) => {
   209→    const res = await fetch(`/api/reviews?purchaseId=${p.id}`)
   210→    if (!res.ok) return false
   211→    return ((await res.json()).reviews || []).length > 0
   212→  }
   213→
   214→  return (
   215→    <div className="homy-page">
   216→      <header className="homy-page-head">
   217→        <p className="homy-eyebrow">Materiales</p>
   218→        <h1 className="homy-page-title mt-1.5">Comprá insumos directo a proveedores</h1>
   219→        <p className="homy-page-sub">
   220→          Buscá lo que necesitás — con o sin obra de por medio — y comprá a la ferretería, corralón o casa
   221→          de tu rubro que mejor te convenga: precio, stock real, reseñas y distancia de cada uno.
   222→        </p>
   223→      </header>
   224→
   225→      {/* tabs buscar / compras */}
   226→      <div role="tablist" aria-label="Secciones de materiales" className="mb-5 flex gap-1 rounded-full bg-white/[0.06] ring-1 ring-[#0A2540]/10 p-1 w-fit max-w-full overflow-x-auto no-scrollbar">
   227→        {(['buscar', 'compras'] as const).map((t) => (
   228→          <button
   229→            key={t}
   230→            role="tab"
   231→            aria-selected={tab === t}
   232→            onClick={() => { setTab(t); navigate(t === 'compras' ? '/panel/cliente/materiales?tab=compras' : '/panel/cliente/materiales', { replace: true }) }}
   233→            className="homy-tab shrink-0"
   234→          >
   235→            {t === 'buscar' ? <Search className="size-4" aria-hidden /> : <ShoppingBag className="size-4" aria-hidden />}
   236→            {t === 'buscar' ? 'Buscar materiales' : 'Mis compras'}
   237→            {t === 'compras' && purchases && purchases.some((p) => ['entregado', 'solicitado'].includes(p.status)) && (
   238→              <span className="homy-badge-pop ml-1 grid size-[18px] place-items-center rounded-full bg-[#FF5A1F] text-[10px] font-extrabold text-white">
   239→                {purchases.filter((p) => ['entregado', 'solicitado'].includes(p.status)).length}
   240→              </span>
   241→            )}
   242→          </button>
   243→        ))}
   244→      </div>
   245→
   246→      {tab === 'buscar' ? (
   247→        <>
   248→          {/* buscador */}
   249→          <div className="relative mb-3">
   250→            <Search className="pointer-events-none absolute left-4 top-1/2 size-4 -translate-y-1/2 text-slate-400" aria-hidden />
   251→            <input
   252→              value={q}
   253→              onChange={(e) => setQ(e.target.value)}
   254→              placeholder="¿Qué necesitás comprar? ej: cemento, membrana, cables…"
   255→              aria-label="Buscar materiales"
   256→              className="homy-glass-input w-full rounded-full py-3.5 pl-11 pr-10 text-[15px]"
   257→            />
   258→            {q && (
   259→              <button onClick={() => setQ('')} aria-label="Limpiar búsqueda" className="absolute right-2 top-1/2 grid size-9 -translate-y-1/2 place-items-center rounded-full text-slate-400 hover:bg-white/70 hover:text-[#0A2540] transition">
   260→                <X className="size-4" aria-hidden />
   261→              </button>
   262→            )}
   263→          </div>
   264→          {/* categorías */}
   265→          <div className="mb-6 flex gap-1.5 overflow-x-auto no-scrollbar py-0.5">
   266→            {CATEGORY_TABS.map((c) => (
   267→              <button key={c.slug} onClick={() => setCat(c.slug)} aria-pressed={cat === c.slug} className="homy-tab shrink-0 min-h-[38px]">
   268→                {c.name}
   269→              </button>
   270→            ))}
   271→          </div>
   272→
   273→          {loading && !results ? (
   274→            <Loading text="Buscando ofertas…" />
   275→          ) : results && results.length === 0 ? (
   276→            <EmptyState
   277→              icon={<Package />}
   278→              title={searched ? 'No encontramos ese producto en stock' : 'Buscá lo que necesitás'}
   279→              hint={searched ? 'Probá con otra palabra (ej: “cemento” en vez de “Loma Negra 50kg”) o mirá por categoría.' : 'Escribí el insumo que necesitás y te mostramos todas las ofertas de los proveedores de la comunidad.'}
   280→            />
   281→          ) : (
   282→            <div className="space-y-4">
   283→              {results?.map((r) => (
   284→                <article key={r.elementId} className="homy-glass homy-lift rounded-3xl p-4 sm:p-5">
   285→                  {/* elemento + explicación */}
   286→                  <div className="flex flex-wrap items-start justify-between gap-2">
   287→                    <div className="min-w-0">
   288→                      <h3 className="text-[17px] font-extrabold text-[#0A2540] leading-snug">{r.name}</h3>
   289→                      <p className="mt-0.5 text-[11px] font-bold uppercase tracking-wider text-slate-400">{r.categoryName} · se vende por {r.unit}</p>
   290→                    </div>
   291→                    {r.hasPro && (
   292→                      <span className="shrink-0 rounded-full bg-gradient-to-r from-[#FFC700] to-[#ffd84d] px-2.5 py-1 text-[10px] font-extrabold uppercase tracking-widest text-[#6b4d00] shadow-sm">
   293→                        ★ Recomendado
   294→                      </span>
   295→                    )}
   296→                  </div>
   297→                  {r.description && <p className="mt-1.5 text-[13.5px] leading-relaxed text-slate-500">{r.description}</p>}
   298→
   299→                  {/* ofertas por proveedor */}
   300→                  {r.offers.length === 0 ? (
   301→                    <p className="mt-3 rounded-2xl bg-[#0A2540]/4 px-4 py-3 text-[13px] text-slate-500">
   302→                      Ahora no hay stock publicado de este elemento — volvé a probar más tarde o mirá elementos similares.
   303→                    </p>
   304→                  ) : (
   305→                    <ul className="mt-3 space-y-2.5">
   306→                      {r.offers.map((o) => (
   307→                        <li key={o.stockId} className={`rounded-2xl p-3.5 ring-1 transition ${o.planPro ? 'bg-[#FFC700]/8 ring-[#FFC700]/45' : 'bg-[#0A2540]/3 ring-[#0A2540]/8'}`}>
   308→                          <div className="flex flex-wrap items-start justify-between gap-3">
   309→                            <div className="flex min-w-0 flex-1 items-start gap-3">
   310→                              <UAvatar name={o.businessName} url={o.providerAvatar} size={40} />
   311→                              <div className="min-w-0">
   312→                                <p className="flex flex-wrap items-center gap-x-1.5 gap-y-0.5 text-sm font-extrabold text-[#0A2540] leading-snug">
   313→                                  {o.businessName}
   314→                                  <VerifyBadge status={o.providerVerified ? 'verificado' : 'none'} />
   315→                                  {o.planPro && (
   316→                                    <span className="rounded-full bg-gradient-to-r from-[#FFC700] to-[#ffd84d] px-2 py-0.5 text-[9px] font-extrabold uppercase tracking-widest text-[#6b4d00]">Recomendado</span>
   317→                                  )}
   318→                                </p>
   319→                                <p className="mt-0.5 flex flex-wrap items-center gap-x-2 gap-y-0.5 text-xs text-slate-500">
   320→                                  <span className="inline-flex items-center gap-1"><MapPin className="size-3 shrink-0 text-[#FF5A1F]" aria-hidden />{o.providerCity || 'Argentina'}{o.distanceKm != null ? ` · ${formatDistance(o.distanceKm)}` : ''}</span>
   321→                                  <span className="inline-flex items-center gap-1"><Store className="size-3 shrink-0 text-slate-400" aria-hidden />{o.providerReviews} reseñas</span>
   322→                                </p>
   323→                                <div className="mt-1 flex items-center gap-1.5">
   324→                                  <UStars rating={o.providerRating} size="text-xs" />
   325→                                  <span className="text-xs font-bold text-slate-600 tabular-nums">{o.providerRating > 0 ? o.providerRating : '—'}</span>
   326→                                </div>
   327→                              </div>
   328→                            </div>
   329→                            <div className="text-right">
   330→                              <p className="homy-num-adapt text-[17px] font-extrabold text-[#0A2540] tabular-nums">{formatARS(o.price)}</p>
   331→                              <p className="text-[11px] font-bold text-slate-400">por {r.unit} · stock: {o.quantity}</p>
   332→                            </div>
   333→                          </div>
   334→                          <div className="mt-3 flex flex-wrap gap-2">
   335→                            <button
   336→                              onClick={() => { setDlg({ offer: o, elementName: r.name, unit: r.unit }); setDlgQty('1'); setDlgNote('') }}
   337→                              className="homy-btn-primary inline-flex min-h-[40px] items-center gap-1.5 rounded-full px-4 text-xs"
   338→                            >
   339→                              <ShoppingBag className="size-3.5" aria-hidden /> Pedir este producto
   340→                            </button>
   341→                            <button
   342→                              onClick={() => navigate(`/mensajes?c=nuevo:${o.providerUserId}`)}
   343→                              className="homy-glass-soft inline-flex min-h-[40px] items-center gap-1.5 rounded-full px-4 text-xs font-bold text-[#1D63B8] hover:bg-white transition"
   344→                            >
   345→                              <MessageCircle className="size-3.5" aria-hidden /> Preguntarle
   346→                            </button>
   347→                          </div>
   348→                        </li>
   349→                      ))}
   350→                    </ul>
   351→                  )}
   352→                </article>
   353→              ))}
   354→            </div>
   355→          )}
   356→        </>
   357→      ) : (
   358→        /* ── MIS COMPRAS ── */
   359→        purchases === null ? (
   360→          <Loading text="Cargando tus compras…" />
   361→        ) : purchases.length === 0 ? (
   362→          <EmptyState
   363→            icon={<ShoppingBag />}
   364→            title="Todavía no hiciste compras"
   365→            hint="Cuando pidas un insumo desde Buscar materiales, acá vas a seguir el pedido hasta el pago y la reseña."
   366→          />
   367→        ) : (
   368→          <div className="space-y-3">
   369→            <p className="homy-page-sub -mt-2">Así funciona: pedís → el proveedor acepta y entrega → pagás (Mercado Pago o efectivo) → calificás tu compra. Todo queda registrado.</p>
   370→            {purchases.map((p) => (
   371→              <PurchaseCard
   372→                key={p.id}
   373→                p={p}
   374→                busy={busy}
   375→                onCancel={() => cancelar(p)}
   376→                onPay={(m) => pagarCharge(p, m)}
   377→                onReview={() => { setRv(p); setRvRating(0); setRvComment('') }}
   378→                hasReview={hasReseña}
   379→              />
   380→            ))}
   381→          </div>
   382→        )
   383→      )}
   384→
   385→      {/* ── diálogo: pedir producto ── */}
   386→      {dlg && (
   387→        <div className="fixed inset-0 z-50 flex items-end justify-center bg-[#0A2540]/45 p-0 backdrop-blur-sm sm:items-center sm:p-6" role="dialog" aria-modal="true" aria-label="Pedir producto">
   388→          <form onSubmit={pedir} className="homy-glass-strong w-full max-w-md rounded-t-3xl p-5 sm:rounded-3xl sm:p-6">
   389→            <div className="flex items-start justify-between gap-3">
   390→              <div>
   391→                <p className="homy-eyebrow">Pedir producto</p>
   392→                <h3 className="mt-0.5 text-lg font-extrabold text-[#0A2540]">{dlg.elementName}</h3>
   393→                <p className="text-[13px] text-slate-500">{dlg.offer.businessName} · {formatARS(dlg.offer.price)} por {dlg.unit}</p>
   394→              </div>
   395→              <button type="button" onClick={() => setDlg(null)} aria-label="Cerrar" className="grid size-9 place-items-center rounded-full text-slate-400 hover:bg-white/70 hover:text-[#0A2540] transition">
   396→                <X className="size-4" aria-hidden />
   397→              </button>
   398→            </div>
   399→            <label className="mt-4 block text-xs font-extrabold uppercase tracking-wider text-slate-400" htmlFor="ped-cant">
   400→              Cantidad ({dlg.unit})
   401→            </label>
   402→            <input
   403→              id="ped-cant"
   404→              type="number" min="0.5" step="0.5" inputMode="decimal"
   405→              value={dlgQty}
