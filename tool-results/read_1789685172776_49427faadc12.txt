    91→/* ------------------------------------------------------------------ */
    92→function useTypewriter(active: boolean) {
    93→  const [text, setText] = useState("");
    94→  const stateRef = useRef({ phrase: 0, char: 0, deleting: false });
    95→
    96→  useEffect(() => {
    97→    if (!active) return;
    98→    const tick = () => {
    99→      const s = stateRef.current;
   100→      const phrase = TYPE_PHRASES[s.phrase % TYPE_PHRASES.length];
   101→      if (!s.deleting) {
   102→        s.char += 1;
   103→        if (s.char >= phrase.length) {
   104→          s.deleting = true;
   105→          return 1700;
   106→        }
   107→        return 48;
   108→      }
   109→      s.char -= 1;
   110→      if (s.char <= 0) {
   111→        s.deleting = false;
   112→        s.phrase += 1;
   113→        return 420;
   114→      }
   115→      return 26;
   116→    };
   117→
   118→    let timeout: ReturnType<typeof setTimeout>;
   119→    const loop = () => {
   120→      const delay = tick();
   121→      const s = stateRef.current;
   122→      setText(
   123→        TYPE_PHRASES[s.phrase % TYPE_PHRASES.length].slice(0, s.char)
   124→      );
   125→      timeout = setTimeout(loop, delay);
   126→    };
   127→    loop();
   128→    return () => clearTimeout(timeout);
   129→  }, [active]);
   130→
   131→  return text;
   132→}

   157→      const trimmed = question.trim();
   158→      if (trimmed.length < 4) return;
   159→
   160→      setPhase("thinking");

   165→        const res = await fetch("/api/homy/agent", {
   166→          method: "POST",
   167→          headers: { "Content-Type": "application/json" },
   168→          body: JSON.stringify({
   169→            message: trimmed,
   170→            mode: "auto",
   171→            lat: lat ?? null,
   172→            lng: lng ?? null,
   173→          }),
   174→        });

   176→        if (!res.ok || !data.ok) {
   177→          setReply({
   178→            ok: false,
   179→            message:
   180→              "El superagente tuvo un problema para razonar tu pedido. Probá de nuevo en unos segundos que sigo acá.",
   181→            suggestions: [],
   182→          });
   183→          setDegraded(true);
   184→        } else {
   185→          setReply(data);
   186→        }
   187→      } catch {
   188→        setReply({
   189→          ok: false,
   190→          message:
   191→            "No pude conectarme con el motor en este momento. Revisá tu conexión y volvé a intentar: sigo acá, cuidando tu hogar.",
   192→          suggestions: [],
   193→        });
   194→        setDegraded(true);
   195→      } finally {
   196→        setPhase("done");
   197→        setValue("");
   198→        inputRef.current?.blur();
   199→        setFlight((f) => (f ? { ...f, stage: "out" } : null));
   200→        if (happyTimer.current) clearTimeout(happyTimer.current);
   201→        happyTimer.current = setTimeout(() => setPhase("idle"), 1900);
   202→      }
   203→    },
   204→    []
   205→  );
   206→
   207→  const onChange = (next: string) => {
   208→    setValue(next);
   209→    if (debounceRef.current) clearTimeout(debounceRef.current);
   210→    if (next.trim().length >= 8) {
   211→      debounceRef.current = setTimeout(
   212→        () => void askHomy(next, location.lat, location.lng),
   213→        1400
   214→      );
   215→    }
   216→  };
   217→
   218→  const onSubmit = (e: FormEvent) => {
   219→    e.preventDefault();

   222→  };
   223→
   224→  useEffect(
   225→    () => () => {
   226→      if (debounceRef.current) clearTimeout(debounceRef.current);

   228→      setFlight(null);
   229→    },
   230→    []
   231→  );
   232→
   233→  // prefers-reduced-motion: la demo ambiental no corre si el usuario la pide
   234→  useEffect(() => {
   235→    const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
   236→    setReducedMotion(mq.matches);
   237→    const onChange = (e: MediaQueryListEvent) => setReducedMotion(e.matches);
   238→    mq.addEventListener("change", onChange);
   239→    return () => mq.removeEventListener("change", onChange);
   240→  }, []);

   244→

   260→          setDemo({ kind: "q", stage: "in", q });
   261→          await wait(DEMO_T.qIn);
   262→          if (!alive) return;
   263→          setDemo({ kind: "q", stage: "think", q });
   264→          await wait(DEMO_T.qThink);
   265→          if (!alive) return;

   275→          setDemo({ kind: "a", stage: "out", q, a });

   299→  const busy = phase === "thinking";

   301→  return (

   306→      <div className="relative z-10 mx-auto w-full max-w-xl">
   307→        <div
   308→          aria-hidden

   314→        {/* Barra premium */}

   328→            aria-hidden
   329→          />
   330→          <input
   331→            ref={inputRef}
   332→            type="text"
   333→            value={value}
   334→            onChange={(e) => onChange(e.target.value)}
   335→            onFocus={() => setFocused(true)}
