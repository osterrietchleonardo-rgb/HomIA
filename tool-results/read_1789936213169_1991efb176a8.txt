    64→export default function SearchScreen({ embedded = false }: { embedded?: boolean }) {
    65→  // pantalla de búsqueda dual con superagente (re-render intencional)
    66→  const route = useRoute()
    67→  const { user, refresh } = useSession()
    68→  const location = useLocation()
    69→
    70→  const [mode, setMode] = useState<Mode>(route.query.mode === 'profesional' ? 'profesional' : 'cliente')
    71→  const [query, setQuery] = useState(route.query.q || '')
    72→  const [cat, setCat] = useState(route.query.cat || '')
    73→  const [pros, setPros] = useState<ProResult[]>([])
    74→  const [jobs, setJobs] = useState<JobResult[]>([])
    75→  const [materials, setMaterials] = useState<MaterialResult[]>([])
    76→  const [comparables, setComparables] = useState<MaterialResult[]>([])
    77→  const [elements, setElements] = useState<ElementResult[]>([])
    78→  const [loading, setLoading] = useState(true)
    79→  const [aiBusy, setAiBusy] = useState(false)
    80→  const [aiState, setAiState] = useState<'idle' | 'listening' | 'thinking' | 'happy'>('idle')
    81→  const [aiMessage, setAiMessage] = useState<string | null>(null)
    82→  const [question, setQuestion] = useState<{ pregunta: string; opciones: string[] } | null>(null)
    83→  const [suggestions, setSuggestions] = useState<string[]>([])
    84→  const [sessionId, setSessionId] = useState<string | null>(null)
    85→  const [steps, setSteps] = useState<{ thought: string; action: string; found: number }[]>([])
    86→  const [showMap, setShowMap] = useState(true)
    87→  const convRef = useRef<{ role: 'user' | 'homy'; content: string }[]>([])
    88→
    89→  useEffect(() => { refresh() /* sync sesión */ }, [refresh])
    90→
    91→  // Búsqueda directa (sin IA) — lista inmediata
    92→  const directSearch = useCallback(async (q: string, category: string) => {
    93→    setLoading(true)
    94→    try {
    95→      const params = new URLSearchParams({ mode, q, cat: category })
    96→      if (location.lat && location.lng) {
    97→        params.set('lat', String(location.lat))
    98→        params.set('lng', String(location.lng))
    99→        params.set('radius', String(location.radiusKm))
   100→      }
   101→      const res = await fetch(`/api/search?${params}`)
   102→      const data = await res.json()
   103→      setPros(data.professionals || [])
   104→      setJobs(data.jobs || [])
   105→      setMaterials(data.materials || [])
   106→      setComparables([])
   107→    } finally {
   108→      setLoading(false)
   109→    }
   110→  }, [mode, location.lat, location.lng, location.radiusKm])
   111→
   112→  useEffect(() => {
   113→    directSearch(query, cat)

   116→  // Superagente: interpreta, pregunta y busca con herramientas reales
   117→  async function askAgent(text: string) {
   118→    if (!text.trim()) return
   119→    setAiBusy(true)
   120→    setAiState('thinking')
   121→    setQuestion(null)
   122→    try {
   123→      const res = await fetch('/api/homy/agent', {
   124→        method: 'POST',
   125→        headers: { 'Content-Type': 'application/json' },
   126→        body: JSON.stringify({
   127→          message: text, mode,
   128→          ...(sessionId ? { sessionId } : {}),
   129→          lat: location.lat ?? null, lng: location.lng ?? null,
   130→        }),
   131→      })
   132→      const data: AgentReply = await res.json()
   133→      if (!data.ok) {
   134→        toast.error(data.error || 'El superagente no pudo responder')
   135→        return
   136→      }
   137→      setSessionId((data as unknown as { sessionId?: string }).sessionId || sessionId)
   138→      setAiMessage(data.message)
   139→      setSuggestions(data.suggestions || [])
   140→      setQuestion(data.question || null)
   141→      setSteps(data.steps || [])
   142→      if (data.results) {
   143→        setPros((prev) => data.results!.professionals?.length ? data.results!.professionals : prev)
   144→        setJobs((prev) => data.results!.jobs?.length ? data.results!.jobs : prev)
   145→        setMaterials((prev) => data.results!.materials?.length ? data.results!.materials : prev)
   146→        setComparables(data.results.comparables || [])
   147→        setElements(data.results.elements || [])
   148→      }
   149→      convRef.current.push({ role: 'user', content: text }, { role: 'homy', content: data.message })
   150→      setAiState('happy')
   151→      setTimeout(() => setAiState('idle'), 1800)
   152→    } catch {
   153→      toast.error('Problema de conexión con el superagente')
   154→      setAiState('idle')
   155→    } finally {
   156→      setAiBusy(false)
   157→    }
   158→  }
   159→
   160→  const [mapPins, setMapPins] = useState<MapPin[]>([])

   162→    async function pins() {
   163→      const params = new URLSearchParams({ mode, q: query, cat, radius: String(location.radiusKm) })
   164→      if (location.lat && location.lng) { params.set('lat', String(location.lat)); params.set('lng', String(location.lng)) }
   165→      const res = await fetch(`/api/search/pins?${params}`)
   166→      if (!res.ok) return
   167→      const data = await res.json()
   168→      setMapPins(data.pins || [])
   169→    }
   170→    pins().catch(() => null)
   171→  }, [mode, query, cat, location.lat, location.lng, location.radiusKm])
   172→

   179→  }

   181→  const hasResults = pros.length + jobs.length + materials.length + comparables.length + elements.length > 0

   183→  const questionDuplicatesMessage = !!question && !!aiMessage && question.pregunta.trim().toLowerCase() === aiMessage.trim().toLowerCase()
   184→
   185→  return (
   186→    <div className="min-h-screen">
   187→      {/* Header de búsqueda — vidrio nocturno */}
   188→      <header className={`homy-glass-dark sticky z-40 overflow-hidden ${embedded ? 'top-16' : 'top-0'}`}>
   189→        <span
   190→          aria-hidden
   191→          className="pointer-events-none absolute inset-0"
   192→          style={{
   193→            background:
   194→              'radial-gradient(70% 130% at 88% -30%, rgba(0,196,255,0.16) 0%, transparent 58%), radial-gradient(42% 90% at -5% 130%, rgba(255,90,31,0.12) 0%, transparent 55%)',
   195→          }}
   196→        />
   197→        <div className="relative max-w-7xl mx-auto px-3 sm:px-4 pt-3 pb-2 flex items-center gap-2 sm:gap-3">
   198→          <button
   199→            onClick={() => navigate('/')}
   200→            className="homy-focus rounded-full shrink-0 transition-transform duration-300 hover:scale-105"
   201→            aria-label="Volver a la home"
   202→          >
   203→            <Homy size={36} state={aiState} />

   205→          <div className="flex-1 min-w-0 relative">
   206→            <Search className="absolute left-4 top-1/2 -translate-y-1/2 size-4 text-slate-400 pointer-events-none" aria-hidden />

   209→              onChange={(e) => setQuery(e.target.value)}
   210→              onKeyDown={(e) => { if (e.key === 'Enter') { askAgent(query) } }}
   211→              placeholder={mode === 'cliente' ? '¿Qué necesitás para tu hogar?' : '“cemento 50kg” · “¿qué hay para plomeros?”'}

   213→            />
   214→            {query && (
   215→              <button
   216→                onClick={() => setQuery('')}

   218→                aria-label="Limpiar búsqueda"
   219→              >
   220→                <X className="size-4" aria-hidden />
   221→              </button>

   223→          </div>
   224→          <button
   225→            onClick={() => askAgent(query || aiMessage || '')}
   226→            disabled={aiBusy}
   227→            className="homy-btn-primary homy-focus shrink-0 px-3.5 sm:px-5 py-2.5 sm:py-3 min-h-[44px] text-sm"
   228→          >
   229→            <Sparkles className="size-4" aria-hidden />
   230→            <span className="hidden sm:inline">{aiBusy ? 'Pensando…' : 'Preguntale a Homy'}</span>
   231→          </button>
   232→        </div>
   233→        {/* Modo + categorías: tira scrolleable en mobile, dos zonas en desktop */}
   234→        <div className="relative max-w-7xl mx-auto px-3 sm:px-4 pb-3 flex items-center gap-2 sm:gap-3 overflow-x-auto no-scrollbar sm:overflow-visible">
   235→          <div role="group" aria-label="Modo de búsqueda" className="flex items-center gap-1 rounded-full bg-white/[0.06] ring-1 ring-white/10 p-1 shrink-0">
   236→            {(['cliente', 'profesional'] as Mode[]).map((m) => (
   237→              <button

   239→                onClick={() => setMode(m)}

   249→          <div className="relative shrink-0 sm:flex-1 sm:min-w-0">

   258→                  {c.name}
   259→                </button>
   260→              ))}
   261→            </div>
   262→            {/* fade decorativo: sugiere que hay más chips deslizables (solo desktop) */}
   263→            <div aria-hidden className="pointer-events-none hidden sm:block absolute inset-y-0 right-0 w-12 bg-gradient-to-l from-[#0A2540]/85 to-transparent" />
   264→          </div>
   265→        </div>
   266→      </header>
   267→
   268→      <div className="max-w-7xl mx-auto px-4 py-6 sm:py-8">
   269→        {/* Respuesta del superagente */}
   270→        {(aiMessage || question) && (
   271→          <div className="mb-6 sm:mb-8 rounded-3xl homy-glass homy-lift relative overflow-hidden p-5 sm:p-6">
   272→            <span
   273→              aria-hidden
   274→              className="pointer-events-none absolute inset-x-0 top-0 h-24"
   275→              style={{ background: 'radial-gradient(60% 100% at 50% 0%, rgba(0,196,255,0.12) 0%, transparent 72%)' }}
   276→            />
   277→            <div className="relative flex items-start gap-3 sm:gap-4">
   278→              <Homy size={56} state={aiBusy ? 'thinking' : 'happy'} />
   279→              <div className="flex-1 min-w-0">
   280→                <p className="homy-eyebrow mb-1.5">Superagente Homy</p>
   281→                {aiMessage && <p className="text-[#0A2540] leading-relaxed font-medium">{aiMessage}</p>}
   282→                {question && !questionDuplicatesMessage && (
   283→                  <div className="mt-4">
   284→                    <p className="font-bold text-[#0A2540]">{question.pregunta}</p>
   285→                    <div className="flex flex-wrap gap-2 mt-2.5">
   286→                      {question.opciones.map((op) => (
   287→                        <button
   288→                          key={op}
   289→                          onClick={() => { askAgent(op) }}
   290→                          className="homy-focus rounded-full min-h-[44px] sm:min-h-[40px] border border-[#1D63B8]/30 bg-white/70 px-4 py-2 text-sm font-bold text-[#1D63B8] hover:bg-[#1D63B8] hover:border-[#1D63B8] hover:text-white transition"
   291→                        >
   292→                          {op}
   293→                        </button>
   294→                      ))}
   295→                    </div>
   296→                  </div>

   298→                {suggestions.length > 0 && !question && (

   302→                        key={s}
   303→                        onClick={() => askAgent(s)}
   304→                        className="homy-focus homy-glass-soft rounded-full min-h-[44px] sm:min-h-[38px] px-4 py-2 text-sm font-semibold text-slate-600 hover:text-[#1D63B8] transition"
   305→                      >
   306→                        {s}
   307→                      </button>
   308→                    ))}
   309→                  </div>
   310→                )}
   311→                {steps && steps.length > 0 && (
   312→                  <details className="mt-4">
   313→                    <summary className="homy-focus inline-flex items-center gap-1.5 rounded-lg text-xs font-semibold text-slate-400 cursor-pointer hover:text-slate-600 transition-colors">
   314→                      Ver razonamiento del agente ({steps.length} pasos)
   315→                    </summary>
   316→                    <ol className="mt-2.5 space-y-1.5 text-xs text-slate-500 border-l-2 border-[#00C4FF]/30 pl-3.5">
   317→                      {steps.map((st, i) => (
   318→                        <li key={i}><b className="text-[#0A2540]">{st.action}</b> → {st.found} resultados {st.thought && `· ${st.thought}`}</li>

   320→                    </ol>
   321→                  </details>

   323→              </div>
   324→            </div>

   337→                className="homy-focus rounded-lg text-[#1D63B8] font-bold hover:underline underline-offset-2 text-left"
   338→              >
   339→                Compartir ubicación para ver pines cercanos
   340→              </button>
   341→            )}
   342→          </div>
   343→          <div className="homy-glass-soft flex items-center gap-3 rounded-full px-4 py-2.5">
   344→            <span className="text-[11px] font-bold text-slate-500 uppercase tracking-wider">Radio</span>
   345→            <span className="text-xs font-extrabold text-[#0A2540] tabular-nums min-w-[46px]">{location.radiusKm} km</span>
   346→            <input
   347→              type="range" min={1} max={100} value={location.radiusKm}
   348→              onChange={(e) => location.setRadius(parseInt(e.target.value))}
   349→              className="homy-range w-28 sm:w-44"
   350→              style={{ ['--range-progress' as string]: `${location.radiusKm}%` }}
   351→              aria-label="Diámetro de alcance"
   352→            />
   353→          </div>
   354→        </div>
   355→
   356→        {/* Mapa */}
   357→        <div className="mb-8 sm:mb-10">
   358→          <div className="homy-section-head">
   359→            <h2 className="homy-section-title">
   360→              <span className="homy-icon-chip homy-chip-blue size-9 shrink-0 [&_svg]:size-[18px]" aria-hidden><Compass /></span>
   361→              Mapa de pines
   362→              <span className="homy-pill tabular-nums">{mapPins.length}</span>
   363→            </h2>
   364→            <button onClick={() => setShowMap(!showMap)} aria-pressed={showMap} className="homy-tab min-h-[44px] sm:min-h-[38px]">
   365→              <Compass className="size-4" aria-hidden />
   366→              {showMap ? 'Ocultar mapa' : 'Ver mapa'}
   367→            </button>
   368→          </div>
   369→          {showMap && (
   370→            <div className="relative isolate rounded-2xl overflow-hidden ring-1 ring-[#0A2540]/10 shadow-[0_24px_60px_-32px_rgba(10,37,64,0.45)]">
   371→              <MapView
   372→                className="h-[320px] w-full sm:h-[420px] lg:h-[480px]"
   373→                center={location.lat && location.lng ? { lat: location.lat, lng: location.lng } : null}
   374→                radiusKm={location.radiusKm}
   375→                pins={pinList()}
   376→                onSelect={(pin) => {
   377→                  if (!user && pin.kind !== 'yo') {
   378→                    toast('Registrate para abrir tarjetas', { description: 'Buscar es libre; abrir datos requiere cuenta.', action: { label: 'Crear cuenta', onClick: () => navigate('/registrarse?volver=/buscar') } })

   380→                  }
   381→                  if (pin.href) navigate(pin.href)

   383→              />
   384→            </div>
   385→          )}
   386→        </div>
   387→
   388→        {loading ? (
   389→          <Loading text="Buscando en la base de HomIA…" />
   390→        ) : !hasResults ? (
   391→          <EmptyState
   392→            icon={<SearchX />}
   393→            title="Sin resultados todavía"

   395→          />
   396→        ) : (
   397→          <div className="space-y-9 sm:space-y-10">
   398→            {/* Profesionales */}
   399→            {mode === 'cliente' && (
   400→              <Section title="Profesionales" count={pros.length} icon={<HardHat />} tone="homy-chip-blue">

   410→
   411→            {/* Recomendaciones del catálogo (agente IA interpreta la necesidad) */}
   412→            {elements.length > 0 && (
   413→              <Section title="Recomendaciones del catálogo" count={elements.length} icon={<Sparkles />} tone="homy-chip-ai">
   414→                <p className="homy-glass-soft rounded-2xl px-5 py-3.5 text-[13px] text-slate-600 leading-relaxed mb-4">
   415→                  Te interpretamos la necesidad y te recomendamos los elementos justos del catálogo estándar — cada uno con su explicación de qué es y para qué sirve. Tocá <strong>Ver precios</strong> para traer stock real de los proveedores.
   416→                </p>
   417→                <div className="homy-stagger grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   418→                  {elements.map((el) => (
   419→                    <ElementCard
   420→                      key={el.elementId}
   421→                      el={el}
   422→                      onViewPrices={(name, slug) => { setQuery(name); setCat(slug); directSearch(name, slug) }}
   423→                    />
   424→                  ))}
   425→                </div>
   426→              </Section>
   427→            )}
   428→
   429→            {/* Materiales */}
   430→            {mode === 'profesional' && (
   431→              <Section title="Materiales en proveedores" count={materials.length} icon={<Package />} tone="homy-chip-mint">
   432→                {materials.length === 0 ? (
   433→                  <p className="homy-glass-soft rounded-2xl px-5 py-4 text-sm text-slate-500">Ningún proveedor publica ese elemento todavía.</p>
   434→                ) : (
   435→                  <div className="homy-stagger grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   436→                    {materials.map((m) => <MaterialCard key={m.stockId} m={m} logged={!!user} />)}
   437→                  </div>
   438→                )}
   439→              </Section>
   440→            )}
   441→
   442→            {/* Comparables */}
   443→            {comparables.length > 0 && (
   444→              <Section title="Comparables: mejor precio por elemento" count={comparables.length} icon={<Scale />} tone="homy-chip-gold">
   445→                <div className="homy-stagger grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   446→                  {comparables.map((m) => <MaterialCard key={m.stockId} m={m} logged={!!user} highlight />)}
   447→                </div>
   448→              </Section>
   449→            )}
   450→
   451→            {/* Bolsa de trabajos */}
   452→            <Section title="Trabajos publicados" count={jobs.length} icon={<Briefcase />} tone="homy-chip-orange">
   453→              {jobs.length === 0 ? (
   454→                <p className="homy-glass-soft rounded-2xl px-5 py-4 text-sm text-slate-500">
   455→                  {mode === 'profesional' ? 'No hay trabajos abiertos para ese criterio. Activá notificaciones en tu perfil.' : 'No hay trabajos publicados para esta búsqueda.'}
   456→                </p>
   457→              ) : (
   458→                <div className="homy-stagger grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   459→                  {jobs.map((j) => <JobCard key={j.id} job={j} logged={!!user} />)}
   460→                </div>
   461→              )}
   462→            </Section>
   463→          </div>
   464→        )}
   465→      </div>
   466→    </div>
   467→  )
   468→}
