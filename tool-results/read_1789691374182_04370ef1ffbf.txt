    95→function useTypewriter(active: boolean) {
    96→  const [text, setText] = useState("");
    97→  const stateRef = useRef({ phrase: 0, char: 0, deleting: false });
    98→
    99→  useEffect(() => {
   100→    if (!active) return;
   101→    const tick = () => {
   102→      const s = stateRef.current;
   103→      const phrase = TYPE_PHRASES[s.phrase % TYPE_PHRASES.length];
   104→      if (!s.deleting) {
   105→        s.char += 1;
   106→        if (s.char >= phrase.length) {
   107→          s.deleting = true;
   108→          return 1700;
   109→        }
   110→        return 48;
   111→      }
   112→      s.char -= 1;
   113→      if (s.char <= 0) {
   114→        s.deleting = false;
   115→        s.phrase += 1;
   116→        return 420;
   117→      }
   118→      return 26;
   119→    };
   120→
   121→    let timeout: ReturnType<typeof setTimeout>;
   122→    const loop = () => {
   123→      const delay = tick();
   124→      const s = stateRef.current;
   125→      setText(
   126→        TYPE_PHRASES[s.phrase % TYPE_PHRASES.length].slice(0, s.char)
   127→      );
   128→      timeout = setTimeout(loop, delay);
   129→    };
   130→    loop();
   131→    return () => clearTimeout(timeout);
   132→  }, [active]);
   133→
   134→  return text;
   135→}

   140→export function HeroSearch() {
   141→  const [value, setValue] = useState("");
   142→  const [focused, setFocused] = useState(false);
   143→  const [phase, setPhase] = useState<"idle" | "thinking" | "done">("idle");
   144→  const [reply, setReply] = useState<AgentReply | null>(null);
   145→  const [degraded, setDegraded] = useState(false);
   146→  const [lastQuery, setLastQuery] = useState("");
   147→  // Coreografía del pedido: la consulta entra por la izquierda (in), se acopla
   148→  // al motor mientras piensa (think) y sale hacia la derecha con la respuesta (out)
   149→  const [flight, setFlight] = useState<{ text: string; stage: "in" | "think" | "out" } | null>(null);
   150→  // Demo ambiental (sin interacción del usuario): muestra el razonamiento en loop
   151→  const [demo, setDemo] = useState<DemoFlight | null>(null);
   152→  const [reducedMotion, setReducedMotion] = useState(false);
   153→  // La demo ambiental se frena mientras el mouse está sobre la barra
   154→  const [hovered, setHovered] = useState(false);
   155→  const inputRef = useRef<HTMLInputElement>(null);
   156→  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   157→  const happyTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   158→  const location = useLocation();
   159→
   160→  // Geometría de la escena: la respuesta tiene que posarse afuera de la barra
   161→  // y desvanecerse contra el borde REAL de la página → se mide la barra, el
   162→  // viewport y el ancho de la propia tarjeta.
   163→  const barWrapRef = useRef<HTMLDivElement>(null);
   164→  const respChipRef = useRef<HTMLSpanElement>(null);
   165→  const [geo, setGeo] = useState({ vw: 1280, barW: 576, respW: 300 });
   166→
   167→  useEffect(() => {
   168→    const measure = () => {
   169→      const bar = barWrapRef.current;
   170→      if (!bar) return;
   171→      const vw = window.innerWidth;
   172→      const barW = bar.getBoundingClientRect().width;
   173→      setGeo((g) => (g.vw !== vw || g.barW !== barW ? { ...g, vw, barW } : g));
   174→    };
   175→    measure();
   176→    window.addEventListener("resize", measure);
   177→    return () => window.removeEventListener("resize", measure);
   178→  }, []);
   179→
   180→  // Cuando monta la tarjeta de respuesta, medir su ancho real
   181→  useEffect(() => {
   182→    if (demo?.kind !== "a") return;
   183→    const el = respChipRef.current;
   184→    const bar = barWrapRef.current;
   185→    if (!el || !bar || el.offsetWidth === 0) return;
   186→    const vw = window.innerWidth;
   187→    const barW = bar.getBoundingClientRect().width;
   188→    setGeo((g) =>
   189→      g.respW !== el.offsetWidth || g.vw !== vw || g.barW !== barW
   190→        ? { vw, barW, respW: el.offsetWidth }
   191→        : g
   192→    );
   193→  }, [demo?.kind, demo?.q]);
   194→
   195→  const askHomy = useCallback(
   196→    async (question: string, lat?: number | null, lng?: number | null) => {
   197→      const trimmed = question.trim();
   198→      if (trimmed.length < 4) return;
   199→
   200→      setPhase("thinking");
   201→      setDegraded(false);
   202→      setLastQuery(trimmed);
   203→      setFlight({ text: trimmed, stage: "in" });
   204→      try {
   205→        const res = await fetch("/api/homy/agent", {
   206→          method: "POST",
   207→          headers: { "Content-Type": "application/json" },
   208→          body: JSON.stringify({
   209→            message: trimmed,
   210→            mode: "auto",
   211→            lat: lat ?? null,
   212→            lng: lng ?? null,
   213→          }),
   214→        });
   215→        const data = (await res.json()) as AgentReply;
   216→        if (!res.ok || !data.ok) {
   217→          setReply({
   218→            ok: false,
   219→            message:
   220→              "El superagente tuvo un problema para razonar tu pedido. Probá de nuevo en unos segundos que sigo acá.",
   221→            suggestions: [],
   222→          });
   223→          setDegraded(true);
   224→        } else {
   225→          setReply(data);
   226→        }
   227→      } catch {
   228→        setReply({
   229→          ok: false,
   230→          message:
   231→            "No pude conectarme con el motor en este momento. Revisá tu conexión y volvé a intentar: sigo acá, cuidando tu hogar.",
   232→          suggestions: [],
   233→        });
   234→        setDegraded(true);
   235→      } finally {
   236→        setPhase("done");
   237→        setValue("");
   238→        inputRef.current?.blur();
   239→        setFlight((f) => (f ? { ...f, stage: "out" } : null));
   240→        if (happyTimer.current) clearTimeout(happyTimer.current);
   241→        happyTimer.current = setTimeout(() => setPhase("idle"), 1900);
   242→      }
   243→    },
   244→    []
   245→  );
   246→
   247→  const onChange = (next: string) => {
   248→    setValue(next);
   249→    if (debounceRef.current) clearTimeout(debounceRef.current);
   250→    if (next.trim().length >= 8) {
   251→      debounceRef.current = setTimeout(
   252→        () => void askHomy(next, location.lat, location.lng),
   253→        1400
   254→      );
   255→    }
   256→  };
   257→
   258→  const onSubmit = (e: FormEvent) => {
   259→    e.preventDefault();
   260→    if (debounceRef.current) clearTimeout(debounceRef.current);
   261→    void askHomy(value, location.lat, location.lng);
   262→  };
   263→
   264→  useEffect(
   265→    () => () => {
   266→      if (debounceRef.current) clearTimeout(debounceRef.current);
   267→      if (happyTimer.current) clearTimeout(happyTimer.current);
   268→      setFlight(null);
   269→    },
   270→    []
   271→  );
   272→
   273→  // prefers-reduced-motion: la demo ambiental no corre si el usuario la pide
   274→  useEffect(() => {
   275→    const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
   276→    setReducedMotion(mq.matches);
   277→    const onChange = (e: MediaQueryListEvent) => setReducedMotion(e.matches);
   278→    mq.addEventListener("change", onChange);
   279→    return () => mq.removeEventListener("change", onChange);
   280→  }, []);
   281→
   282→  const ambientAllowed =
   283→    !reducedMotion &&
   284→    phase === "idle" &&
   285→    !focused &&
   286→    !hovered &&
   287→    value === "" &&
   288→    !reply &&
   289→    !flight;
   290→
   291→  // Loop de la demo: consulta (izq → motor) · respuesta (motor → der).
   292→  // Un elemento por vez; cada etapa espera a que la anterior termine.
   293→  useEffect(() => {
   294→    if (!ambientAllowed) return;
   295→    let alive = true;
   296→    const timers: ReturnType<typeof setTimeout>[] = [];
   297→    const wait = (ms: number) =>
   298→      new Promise<void>((res) => {
   299→        const t = setTimeout(res, ms);
   300→        timers.push(t);
   301→      });
   302→    (async () => {
   303→      // Cooldown al (re)iniciar: sin parpadeos cuando el mouse cruza la barra
   304→      await wait(DEMO_T.restart);
   305→      if (!alive) return;
   306→      while (alive) {
   307→        for (let i = 0; i < DEMO_PAIRS.length && alive; i += 1) {
   308→          const { q, a } = DEMO_PAIRS[i];
   309→          // 1) La consulta vuela desde la izquierda y, al llegar, se desvanece
   310→          //    DETRÁS del vidrio de la barra (un solo vuelo continuo)
   311→          setDemo({ kind: "q", q });
   312→          await wait(DEMO_T.qFly);
   313→          if (!alive) return;
   314→          setDemo(null); // respiro: la escena queda vacía un instante (sin pisarse)
   315→          await wait(DEMO_T.gap);
   316→          if (!alive) return;
   317→          // 2) La respuesta surge desde detrás del vidrio hacia afuera de la
   318→          //    barra, se deja leer y se desvanece contra el borde de la página
   319→          setDemo({ kind: "a", stage: "in", q, a });
   320→          await wait(DEMO_T.aIn);
   321→          if (!alive) return;
   322→          setDemo({ kind: "a", stage: "hold", q, a });
   323→          await wait(DEMO_T.aHold);
   324→          if (!alive) return;
   325→          setDemo({ kind: "a", stage: "out", q, a });
   326→          await wait(DEMO_T.aOut + DEMO_T.next);
   327→          if (!alive) return;
   328→        }
   329→      }
   330→    })();
   331→    return () => {
   332→      alive = false;
   333→      timers.forEach(clearTimeout);
   334→      setDemo(null);
   335→    };
   336→  }, [ambientAllowed]);
   337→
   338→  const placeholderText = useTypewriter(!focused && value === "" && phase !== "thinking");
   339→  const demoRunning = demo !== null;
   340→  const showPlaceholder = demoRunning ? "" : placeholderText || "¿Qué necesita tu hogar?";
   341→
   342→  const homyState: HomyState = useMemo(() => {
   343→    if (phase === "thinking") return "thinking";
   344→    if (phase === "done") return "happy";
   345→    if (focused || value.length > 0) return "listening";
   346→    // La demo también le cuenta la historia a la mascota
   347→    if (demo?.kind === "q") return "listening";
   348→    if (demo?.kind === "a") return "happy";
   349→    return "idle";
   350→  }, [phase, focused, value, demo]);
   351→
   352→  const busy = phase === "thinking";
   353→
   354→  return (
   355→    <MotionConfig reducedMotion="user">
   356→    <div className="mx-auto w-full max-w-6xl">
   357→      {/* Homy flotando detrás de la barra: las piernas quedan ocultas
   358→          tras el borde superior (z-0), nunca tapa el texto del input */}
   359→      <div ref={barWrapRef} className="relative z-10 mx-auto w-full max-w-xl">
   360→        <div
   361→          aria-hidden
   362→          className="pointer-events-none relative z-0 mx-auto w-fit origin-bottom animate-float-slow mb-[-26px] sm:mb-[-34px]"
   363→        >
   364→          <Homy size={88} state={homyState} />
   365→        </div>
   366→
   367→        {/* Escenario ambiental DETRÁS del vidrio (z-[5] queda por debajo del
   368→            form z-10): el backdrop-filter de la barra esmerila lo que pasa por
   369→            detrás → la consulta se funde con el vidrio al llegar. */}
   370→        <div
   371→          aria-hidden
   372→          className="pointer-events-none absolute inset-x-0 bottom-0 z-[5] flex h-14 items-center sm:h-16"
   373→        >
   374→          {/* CONSULTA: un solo vuelo — entra desde la izquierda legible y, al
   375→              llegar a la barra, sigue su camino POR DETRÁS del vidrio mientras
   376→              se desvanece (el vidrio la va tragando de a poco) */}
   377→          {demo?.kind === "q" && (
   378→            <motion.span
   379→              initial={{ opacity: 0, x: -350, scale: 0.88 }}
   380→              animate={{
   381→                opacity: [0, 1, 1, 0],
   382→                x: [-350, -140, -45, 15],
   383→                scale: [0.88, 1, 0.97, 0.93],
   384→              }}
   385→              transition={{
   386→                duration: DEMO_T.qFly / 1000,
   387→                times: [0, 0.32, 0.72, 1],
   388→                ease: "easeInOut",
   389→              }}
   390→              className="homy-flight-chip max-w-[min(62vw,300px)] px-3.5 py-1.5 sm:max-w-[340px]"
   391→            >
   392→              <Sparkles className="size-3.5 shrink-0 text-tech" aria-hidden />
   393→              <span className="min-w-0 truncate text-[13px] font-bold text-navy">
   394→                {demo.q}
   395→              </span>
   396→            </motion.span>
   397→          )}
   398→
   399→          {/* RESPUESTA: arranca escondida detrás del vidrio (x:-44), surge
   400→              afuera de la barra, se deja leer y deriva hasta desvanecerse
   401→              contra el borde real de la página (medido en vivo) */}
   402→          {demo?.kind === "a" &&
   403→            (() => {
   404→              const gapPx = Math.max(0, (geo.vw - geo.barW) / 2);
   405→              const flotante = gapPx < 130; // poco lugar: flota sobre el borde derecho de la barra
   406→              const restX = flotante
   407→                ? gapPx - 14
   408→                : Math.min(24 + geo.respW, gapPx - 16);
   409→              const restY = flotante ? -46 : 0;
   410→              const exitX = gapPx + 40; // el borde derecho termina ~40px pasada la página
   411→              return (
   412→                <motion.span
   413→                  ref={respChipRef}
   414→                  initial={{ opacity: 0, x: -44, y: 0 }}
   415→                  animate={
   416→                    demo.stage === "in"
   417→                      ? { opacity: 1, x: restX, y: restY }
   418→                      : demo.stage === "hold"
   419→                        ? { opacity: 1, x: restX, y: restY, scale: [1, 1.015, 1] }
   420→                        : { opacity: 0, x: exitX, y: restY - 8, scale: 0.97 }
   421→                  }
   422→                  transition={
   423→                    demo.stage === "in"
   424→                      ? { duration: DEMO_T.aIn / 1000, ease: [0.22, 1, 0.36, 1] }
   425→                      : demo.stage === "hold"
   426→                        ? {
   427→                            scale: { duration: 2.7, ease: "easeInOut" },
   428→                            default: { duration: 0.3 },
   429→                          }
   430→                        : { duration: DEMO_T.aOut / 1000, ease: [0.6, 0.05, 0.9, 0.4] }
   431→                  }
   432→                  style={
   433→                    flotante
   434→                      ? undefined
   435→                      : { maxWidth: Math.max(150, Math.min(340, gapPx - 40)) }
   436→                  }
   437→                  className="homy-flight-chip homy-flight-chip--ai ml-auto max-w-[min(58vw,280px)] px-3.5 py-1.5"
   438→                >
   439→                  <BadgeCheck className="size-3.5 shrink-0 text-tech" aria-hidden />
   440→                  <span className="min-w-0 truncate text-[12.5px] font-bold text-navy/85">
   441→                    {demo.a}
   442→                  </span>
   443→                </motion.span>
   444→              );
   445→            })()}
   446→        </div>
   447→
   448→        {/* Barra premium */}
   449→        <form
   450→          onSubmit={onSubmit}
   451→          className={cn(
   452→            "bar-relief relative z-10 flex h-14 items-center gap-2 rounded-full pr-2 pl-5 transition-shadow duration-500 sm:h-16 sm:pl-6",
   453→            "focus-within:shadow-[0_22px_54px_-16px_rgba(0,196,255,0.4),inset_0_2px_8px_rgba(10,37,64,0.05)]"
   454→          )}
   455→          role="search"
   456→          onMouseEnter={() => setHovered(true)}
   457→          onMouseLeave={() => setHovered(false)}
   458→        >
   459→          <Sparkles
   460→            className={cn(
   461→              "size-5 shrink-0 transition-colors duration-300",
   462→              homyState === "idle" ? "text-tech/60" : "text-ai"
   463→            )}
   464→            aria-hidden
   465→          />
   466→          <input
   467→            ref={inputRef}
   468→            type="text"
   469→            value={value}
   470→            onChange={(e) => onChange(e.target.value)}
   471→            onFocus={() => setFocused(true)}
   472→            onBlur={() => setFocused(false)}
   473→            placeholder={showPlaceholder}
   474→            aria-label="Contale a Homy qué necesitás"
   475→            maxLength={280}
   476→            className="h-full min-w-0 flex-1 bg-transparent text-[15px] font-medium text-navy outline-none placeholder:text-navy/35 sm:text-base"
   477→          />
   478→          <button
   479→            type="submit"
   480→            disabled={busy || value.trim().length === 0}
   481→            aria-label="Preguntarle a Homy"
   482→            className="homy-btn-primary size-10 shrink-0 sm:size-11"
   483→          >
   484→            {busy ? (
   485→              <Loader2 className="size-4.5 animate-spin" aria-hidden />
   486→            ) : (
   487→              <ArrowRight className="size-4.5" aria-hidden />
   488→            )}
   489→          </button>
   490→        {/* Vuelo de la consulta REAL (usuario): entra por la izquierda, se acopla al motor
   491→            y sale hacia la derecha cuando llega la respuesta (solo transform/opacity) */}
   492→        {flight && (
