    61→

    83→  const directSearch = useCallback(async (q: string, category: string) => {

    94→      setPros(data.professionals || [])
    95→      setJobs(data.jobs || [])
    96→      setMaterials(data.materials || [])
    97→      setComparables([])
    98→    } finally {
    99→      setLoading(false)
   100→    }

   107→  // Superagente: interpreta, pregunta y busca con herramientas reales
   108→  async function askAgent(text: string) {
   109→    if (!text.trim()) return

   111→    setAiState('thinking')
   112→    setQuestion(null)
   113→    try {
   114→      const res = await fetch('/api/homy/agent', {
   115→        method: 'POST',
   116→        headers: { 'Content-Type': 'application/json' },
   117→        body: JSON.stringify({
   118→          message: text, mode, sessionId,
   119→          lat: location.lat, lng: location.lng,
   120→        }),
   121→      })
   122→      const data: AgentReply = await res.json()
   123→      if (!data.ok) {
   124→        toast.error(data.error || 'El superagente no pudo responder')
   125→        return
   126→      }
   127→      setSessionId((data as unknown as { sessionId?: string }).sessionId || sessionId)
   128→      setAiMessage(data.message)
   129→      setSuggestions(data.suggestions || [])
   130→      setQuestion(data.question || null)
   131→      setSteps(data.steps || [])
   132→      if (data.results) {
   133→        setPros((prev) => data.results!.professionals?.length ? data.results!.professionals : prev)
   134→        setJobs((prev) => data.results!.jobs?.length ? data.results!.jobs : prev)
   135→        setMaterials((prev) => data.results!.materials?.length ? data.results!.materials : prev)
   136→        setComparables(data.results.comparables || [])
   137→      }
   138→      convRef.current.push({ role: 'user', content: text }, { role: 'homy', content: data.message })
   139→      setAiState('happy')
   140→      setTimeout(() => setAiState('idle'), 1800)
   141→    } catch {
   142→      toast.error('Problema de conexión con el superagente')
   143→      setAiState('idle')
   144→    } finally {
   145→      setAiBusy(false)
   146→    }
   147→  }
